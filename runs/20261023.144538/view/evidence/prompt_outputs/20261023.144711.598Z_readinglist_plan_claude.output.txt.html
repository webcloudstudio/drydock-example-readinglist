=== AC books-list-order ===
Intent: Listing returns books in the order they were created.
Suite: scoped

from persistence.testing import db_session

titles = ["Order One", "Order Two", "Order Three"]
with db_session() as db:
    created_ids = [db.books.create(title=t, author="Order Author").id for t in titles]
    listed_ids = [row.id for row in db.books.list()]
    ours = [i for i in listed_ids if i in created_ids]
    assert ours == created_ids
    assert created_ids == sorted(created_ids)
=== END AC books-list-order ===

=== AC books-delete ===
Intent: A deleted book is absent from the list on the next read and other books remain.
Suite: scoped

from persistence.testing import db_session

with db_session() as db:
    first = db.books.create(title="Delete First", author="Delete Author")
    second = db.books.create(title="Delete Second", author="Delete Author")
    db.books.delete(id=first.id)
    listed_ids = [row.id for row in db.books.list()]
    assert first.id not in listed_ids
    assert second.id in listed_ids
=== END AC books-delete ===
