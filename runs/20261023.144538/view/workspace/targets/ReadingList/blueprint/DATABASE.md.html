Programmatic Acceptance

=== AC books-create ===
Intent: Creating a book returns a row carrying the supplied title and author and an assigned id.
Suite: scoped

from persistence.testing import db_session

title = "Create Title" author = "Create Author" with db_session() as db: row = db.books.create(title=title, author=author) assert row.id is not None assert row.title == title assert row.author == author === END AC books-create ===

=== AC books-get ===
Intent: A created book can be read back by id with the supplied title and author.
Suite: scoped

from persistence.testing import db_session

title = "Get Title" author = "Get Author" with db_session() as db: created = db.books.create(title=title, author=author) fetched = db.books.get(id=created.id) assert fetched is not None assert fetched.id == created.id assert fetched.title == title assert fetched.author == author === END AC books-get ===

=== AC books-list-order ===
Intent: Listing returns books in the order they were created.
Suite: scoped

from persistence.testing import db_session

titles = ["Order One", "Order Two", "Order Three"] with db_session() as db: created_ids = [db.books.create(title=t, author="Order Author").id for t in titles] listed_ids = [row.id for row in db.books.list()] ours = [i for i in listed_ids if i in created_ids] assert ours == created_ids assert created_ids == sorted(created_ids) === END AC books-list-order ===

=== AC books-delete ===
Intent: A deleted book is absent from the list on the next read and other books remain.
Suite: scoped

from persistence.testing import db_session

with db_session() as db: first = db.books.create(title="Delete First", author="Delete Author") second = db.books.create(title="Delete Second", author="Delete Author") db.books.delete(id=first.id) listed_ids = [row.id for row in db.books.list()] assert first.id not in listed_ids assert second.id in listed_ids === END AC books-delete ===
