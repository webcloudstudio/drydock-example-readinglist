Programmatic Acceptance

=== AC database-books-create ===
Intent: The books create operation returns the declared book row with the supplied values.

from persistence.testing import db_session

title = "The Left Hand of Darkness" author = "Ursula K. Le Guin" insertion_order = 1

with db_session() as db: row = db.books.create( title=title, author=author, insertion_order=insertion_order, ) assert row.title == title assert row.author == author assert row.insertion_order == insertion_order === END AC database-books-create ===

=== AC database-books-list ===
Intent: The books list operation returns books in insertion order.

from persistence.testing import db_session

first_title = "Dune" first_author = "Frank Herbert" second_title = "Kindred" second_author = "Octavia E. Butler"

with db_session() as db: db.books.create( title=first_title, author=first_author, insertion_order=1, ) db.books.create( title=second_title, author=second_author, insertion_order=2, ) rows = db.books.list()

assert [row.title for row in rows] == [first_title, second_title] assert [row.author for row in rows] == [first_author, second_author] === END AC database-books-list ===

=== AC database-books-delete ===
Intent: The books delete operation returns the selected identifier and removes that book from subsequent reads.

from persistence.testing import db_session

title = "A Wizard of Earthsea" author = "Ursula K. Le Guin"

with db_session() as db: created = db.books.create( title=title, author=author, insertion_order=1, ) deleted = db.books.delete(id=created.id) assert deleted.id == created.id

remaining = db.books.list() assert all(row.id != created.id for row in remaining) === END AC database-books-delete ===
