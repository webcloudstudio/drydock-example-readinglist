=== AC books-get ===

from persistence.testing import db_session

title = "ac-get-title"
author = "ac-get-author"

=== AC books-list ===
Intent: books.list returns books in the order they were added.
Suite: scoped

from persistence.testing import db_session

titles = ["ac-list-first", "ac-list-second", "ac-list-third"]
with db_session() as db:
    created_ids = [db.books.create(title=t, author="ac-list-author").id for t in titles]
    listed_ids = [row.id for row in db.books.list()]
    mine = [i for i in listed_ids if i in created_ids]
    assert mine == created_ids
=== END AC books-list ===

=== AC books-delete ===
Intent: books.delete removes a book so it no longer appears in the next list read, and leaves other books in place.
Suite: scoped

from persistence.testing import db_session

with db_session() as db:
    keep = db.books.create(title="ac-delete-keep", author="ac-delete-author")
    gone = db.books.create(title="ac-delete-gone", author="ac-delete-author")
    db.books.delete(id=gone.id)
    listed_ids = [row.id for row in db.books.list()]
    assert gone.id not in listed_ids
    assert keep.id in listed_ids
=== END AC books-delete ===
