=== AC books-list ===
Intent: books.list returns books in the order they were added.
Suite: scoped

from persistence.testing import db_session

Suite: scoped

from persistence.testing import db_session

with db_session() as db: keep = db.books.create(title="ac-delete-keep", author="ac-delete-author") gone = db.books.create(title="ac-delete-gone", author="ac-delete-author") db.books.delete(id=gone.id) listed_ids = [row.id for row in db.books.list()] assert gone.id not in listed_ids assert keep.id in listed_ids === END AC books-delete ===
